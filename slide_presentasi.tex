\begin{frame}{Latar Belakang \& Konteks Strategis}
    Setiap aset informasi menentukan \textbf{kecepatan keputusan} dan \textbf{mitigasi risiko}.
    \vspace{0.3cm}
    \begin{alertblock}{Tantangan Kondisi \textit{As-Is}}
        Prosedur masih dihambat proses manual, input data berulang, persetujuan lambat, serta hilangnya konsistensi metadata -- menyebabkan risiko ketidakpatuhan (\textit{compliance risk}) dan inefisiensi.
    \end{alertblock}
    \vspace{0.3cm}
    \textbf{Tujuan Pelatihan:}
    \begin{enumerate}
        \item \textbf{Mengevaluasi} prosedur yang ada dengan metrik yang objektif (Kriteria 4.3.1).
        \item \textbf{Merancang} alur kerja baru yang integratif (\textit{To-Be}).
        \item \textbf{Merumuskan} rancangan SOP Digital berorientasi sistem.
    \end{enumerate}
\end{frame}

% SLIDE 3
\begin{frame}{Mengapa Mengadopsi Standar PARBICA (Guideline 13)?}
    Untuk membangun kerangka evaluasi, modul PLN menggunakan alat ukur dari panduan internasional \textbf{PARBICA Toolkit} (afisiliasi resmi \textit{International Council on Archives}).
    \vspace{0.4cm}
    \begin{itemize}
        \item \textbf{Konteks Relevan:} Fokus pada "Masa Transisi" dari kertas analog ke lingkungan kerja digital, sangat spesifik dengan tantangan BUMN.
        \item \textbf{Praktis:} Memberikan diagnosis kesiapan organisasi melalui \textit{Checklist Diagnostik} skala 1-5, tidak sekadar kerangka filosofis.
    \end{itemize}
    \vspace{0.3cm}
    \begin{block}{Pedoman Kunci \textit{Good Governance}}
        Sistem tata kelola kearsipan bukanlah tahap \textit{post-action} administratif, melainkan aktivitas \textbf{By Design} yang tertanam otomatis dalam alur transaksi bisnis.
    \end{block}
\end{frame}

% SLIDE 4
\begin{frame}{3 Prinsip Utama Evaluasi Prosedur}
    Kami menggunakan pijakan operasional berikut pada setiap titik evaluasi:
    \vspace{0.4cm}
    
    \begin{description}[Process-System]
        \item[\textcolor{plnblue}{1. Value-Added Focus}] \hfill \\ Setiap tahapan memegang memori (\textit{capture}), index, atau routing wajib dipertimbangkan nilai tambahnya bagi kecepatan layanan/kepatuhan.
        
        \vspace{0.2cm}
        \item[\textcolor{plnblue}{2. Single Source of Truth}] \hfill \\ Data dientri \textbf{satu kali} pada sistem hulu dan disalurkan secara otonom melintasi repositori arsip.
        
        \vspace{0.2cm}
        \item[\textcolor{plnorange}{3. Process-System Alignment}] \hfill \\ Prosedur dibuat untuk memanfaatkan kekuatan maksimal aplikasi yang ada (bukan sekadar membuat ``solusi alternatif manual'').
    \end{description}
\end{frame}

% SLIDE 5
\begin{frame}{Tahap 1: Pengumpulan Fakta Lapangan (\textit{Fact-Finding})}
    Menghindari \textit{redesign} prosedur kearsipan dari teori semata. Gunakan 3 teknik praktis di lapangan:
    \vspace{0.4cm}
    
    \begin{columns}[T]
        \column{0.33\textwidth}
        \textbf{1. Document Shadowing}\\
        \small Melacak posisi absolut dari sebuah dokumen dari ia terbit, disetujui, hingga diletakkan pada repositori (Cari \textit{titik buntu/bottle-neck}).
        
        \column{0.33\textwidth}
        \textbf{2. Ekstraksi Log Sistem}\\
        \small Unduh riwayat persetujuan dari aplikasi, ukur kapan SLA meleset pada jam riil. Data empiris berdasar \textit{time-stamp}.
        
        \column{0.33\textwidth}
        \textbf{3. Gemba Walk}\\
        \small Lakukan interview acak pada \textit{user} garis tepi untuk menemukan anomali lapangan (seperti koneksi yang memaksa cetak-scan kertas).
    \end{columns}
\end{frame}

\begin{frame}{Tahap 3: Pembuatan Desain Terintegrasi (To-Be)}
    Matriks \textit{As-Is} kini dikonversi menjadi alur BPMN terotomasi menggunakan parameter \textbf{Pemicu Otomatis} (\textit{Trigger}):
    \vspace{0.3cm}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Alur Manual (\textit{Push}):}
        \begin{itemize}
            \small
            \item Aktivitas diisi teknisi.
            \item Selesai, teknisi cetak log PDF.
            \item Manajer menandatangani.
            \item Sekretariat unggah manual ke \textit{Folder Bersama}.
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Alur Terintegrasi (\textit{Event-Pull}):}
        \begin{itemize}
            \small
            \item Transaksi berstatus \texttt{TECO} (Selesai).
            \item Sistem otomatis \textit{Auto-Capture} draft arsip.
            \item Routing \& Validasi 1x24 jam by E-Office.
            \item Arsip final menempati \textit{Storage} dgn metadata valid.
        \end{itemize}
    \end{columns}
\end{frame}

% SLIDE 8
\begin{frame}{Tahap 4: Merumuskan SOP Digital PLN Terpadu}
    SOP Konvensional perlu dirubah. SOP Kearsipan Modern mengikat \textbf{Integrasi Manusia + Sistem IT}.
    \vspace{0.3cm}

    \textbf{Komponen Konstruksi SOP Digital:}
    \vspace{0.1cm}
    \begin{enumerate}
        \small
        \item \textbf{Pemicu Penciptaan (\textit{Trigger Point}):} Aksi sistemik mana yang memulai \textit{penarikan data} arsip.
        \item \textbf{Mapping Metadata Mandatory:} Atribut ID (mis. No Aset, Tanggal) dikunci.
        \item \textbf{Peran Digital (\textit{Digital Custodian}):} Penanggungjawab persetujuan sistem.
        \item \textbf{Aturan SLA Validasi:} Batas Waktu \& Eskalasi Bypass Otomatis.
        \item \textbf{Exception Handling \& Fallback:} Bila Server/API Kearsipan terputus (>4 jam).
        \item \textbf{Siklus Retensi Otomatis:} Durasi penyimpanan sebelum JRA diberlakukan.
    \end{enumerate}
\end{frame}

% SLIDE 9
\begin{frame}{Tahap 5: Kalkulasi KPI \& ROI Implementasi}
    Evaluasi akhir dari perombakan prosedur direkam lewat Key Performance Indicators (KPI):
    \vspace{0.3cm}
    
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Pengukuran (Tracker Metric):}
        \small
        \begin{itemize}
            \item \textbf{Waktu Siklus (\textit{Cycle Time}):} Hari tunggu proses.
            \item \textbf{Tingkat Kepatuhan Prosedur (\textit{Compliance Rate}):} Tingkat tertib kepatuhan mengisi log arsip SPK.
            \item \textbf{Waktu Temu Kembali (\textit{Retrieval Speed}):} Hitungan menit mencari bukti persetujuan saat di-audit.
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Financial Business-Case:}
        \small
        \begin{itemize}
            \item \textbf{Biaya Operasional Kertas:} Nihil.
            \item \textbf{Manajemen Waktu Man-Hours:} Bisa di-salurkan pada mitigasi lain.
            \item \textbf{Penurunan Penalti Audit:} Rekaman 100\% ada, dan otentik (\textit{hash} aman).
        \end{itemize}
    \end{columns}
\end{frame}
